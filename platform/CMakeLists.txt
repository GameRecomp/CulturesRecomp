set(_use_sdl3 OFF)
if(RC_WITH_SDL3 STREQUAL "ON")
  find_package(SDL3 CONFIG REQUIRED COMPONENTS SDL3)
  set(_use_sdl3 ON)
elseif(RC_WITH_SDL3 STREQUAL "AUTO")
  find_package(SDL3 CONFIG QUIET COMPONENTS SDL3)
  if(SDL3_FOUND)
    set(_use_sdl3 ON)
  endif()
endif()

if(_use_sdl3)
  message(STATUS "platform: SDL3 ${SDL3_VERSION}")
  add_library(rc_platform STATIC src/platform_sdl3.c)
  target_link_libraries(rc_platform PUBLIC SDL3::SDL3)
else()
  message(STATUS "platform: null backend (no SDL3)")
  add_library(rc_platform STATIC src/platform_null.c)
endif()
target_include_directories(rc_platform PUBLIC include)
target_link_libraries(rc_platform PRIVATE rc_warnings)

if(RC_BUILD_TESTS)
  add_executable(test_platform tests/test_platform.c)
  target_include_directories(test_platform PRIVATE "${PROJECT_SOURCE_DIR}/runtime/tests")
  target_link_libraries(test_platform PRIVATE rc_platform rc_warnings)
  add_test(NAME platform COMMAND test_platform)
  set_tests_properties(platform PROPERTIES ENVIRONMENT "SDL_VIDEO_DRIVER=dummy")
endif()
